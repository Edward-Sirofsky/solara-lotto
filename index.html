function waitForProvider(providerUrl, timeoutMs = 3000) { return new Promise((resolve, reject) => { const startTime = Date.now(); const timer = setTimeout(() => { resolve(false); // Timeout reached }, timeoutMs); fetch(providerUrl) .then(response => { clearTimeout(timer); // Clear the timeout if the request succeeds if (response.ok) { resolve(true); // Request succeeded } else { resolve(false); // Request failed with a non-200 status code } }) .catch(error => { clearTimeout(timer); // Clear the timeout if there's an error console.error('Request to provider failed:', error); resolve(false); // Request failed due to error }); }); }
